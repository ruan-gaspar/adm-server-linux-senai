sudo iptables -A INPUT -p tcp --dport 80 -j DROP          "BLOCK PORTA 80"

sudo iptables -D INPUT -p tcp --dport 80 -j DROP	  "LIBERA A PORTA 80 NOVAMENTE"

sudo iptables -L INPUT -n --line-numbers                  "LISTA AS REGRAS ATIVAS"
